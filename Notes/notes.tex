\subsection{あいさつ （挨拶）- Aisatsu}
\begin{itemize}
    \item おはよう。\text{Good morning.}
    \item おはようございます。\text{Good morning. (Polite)}
    \item こんにちは。\text{Good afternoon.}
    \item こんばんは。\text{Good evening.}
    \item さようなら。\text{Goodbye.}
    \item おやすみ（なさい）。\text{Good night.}
    \item ありがとう。\text{Thank you.}
    \item ありがとうございます。\text{Thank you. (Polite)}
    \item すみません。\text{Excuse me.; I'm sorry.}
    \item いいえ。\text{No.; Not at all.}
    \item いってきます。；行ってきます。\text{I'll go and come back.}
    \item いってらしゃい。；行ってらしゃい。\text{Please go and come back.}
    \item ただいま。\text{I'm home.}
    \item おかえり（なさい）。\text{Welcome home.}
    \item いただきます。\text{Thank you for the meal. (Before eating)}
    \item ごちそうさま（でした）。\text{Thank you for the meal. (After eating)}
    \item はじめまして。\text{How do you do?}
    \item 〜です。\text{I am...}
    \item よろしくおねがいします。\text{Nice to meet you.}
\end{itemize}

\subsection{じゅぎょうにべんりなひょうげん - Useful Classroom Expressions}
\noindent Teacher Instructions:
\begin{enumerate}
    \item きいてください。\text{Please listen or Please ask.}
    \item いってください。\text{Please speak.}
    \item ＿をかいてください。\text{Please write \textunderscore .}
    \item よんでください。 \text{Please read.}
    \item リピートしてください。\text{Please repeat (after me).}
    \item しゅくだいください。\text{Homework please.}
    \item わかりましたか？\text{Do you understand?}
    \item ＿ぺージをみてください。\text{Please look at page \textunderscore .}
\end{enumerate}

\noindent Both:
\begin{enumerate}
    \item もういちどいってください。\text{Please say it again.}
    \item ちょっとまってください。\text{Please wait a moment.}
\end{enumerate}

\noindent Student Questions/Requests:
\begin{enumerate}
    \item はい、わかりました。\text{Yes, I understand or Yes, I understood.}
    \item いいえ、わかりません。\text{No, I don't understand or No, I don't know.}
    \item ゆっくりいってください。\text{Please say it slowly.}
    \item ＿はなんですか。\text{What is this \textunderscore ?}
    \item えいごではなしてもいいですか？\text{May I speak in English?}
    \item ＿はにほんごでなんですか？\text{What is \textunderscore in Japanese?}
    \item ＿はえいごでなんですか？\text{What is \textunderscore in English?}
\end{enumerate}
